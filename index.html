async function sendTestMessage() { // 先檢查有沒有登入，沒登入什麼都不能做 if (!liff.isLoggedIn()) { alert('請先點擊登入喔！'); return; } // 準備要發送的訊息內容 const messagePayload = [ { type: 'text', text: '大家好！這是我從測試小程式發送的訊息 🚀' } ]; // 情況 A：如果在手機 LINE App 聊天室內打開 if (liff.isInClient()) { try { await liff.sendMessages(messagePayload); alert('在 LINE 聊天室中發送成功！'); } catch (error) { alert('LINE 內發送失敗：' + JSON.stringify(error)); } } // 情況 B：如果是在一般瀏覽器（Safari/Chrome/電腦版）打開 else { // 檢查後台有沒有開功能 if (liff.isApiAvailable('shareTargetPicker')) { try { // 叫出 LINE 好友/群組選擇器 const result = await liff.shareTargetPicker(messagePayload); if (result) { alert('分享成功！訊息已傳送給指定好友。'); } else { alert('你取消了分享。'); } } catch (error) { alert('瀏覽器分享失敗，請確認 LINE 後台的 shareTargetPicker 開關是否有打開：' + JSON.stringify(error)); } } else { alert('此 LIFF 尚未在後台啟用 shareTargetPicker 功能。'); } } }
